⭐ Notes on the 8086 mapping
✔ Endianness is perfect
All immediates and addresses (ll hh) are little‑endian, exactly like EX716.

✔ Stack semantics match
8086 stack grows downward

PUSH/POP store words little‑endian

TOS = [SP]

This matches EX716 exactly.

✔ Memory layout matches
EX716’s 16‑bit address space maps directly into 8086’s 20‑bit space.

✔ Flags match closely
8086 Z, S, C, O flags map cleanly to EX716 Z, N, C, O.

✔ CAST/POLL map cleanly to INT nn
You can assign:

CAST → INT 30h

POLL → INT 31h

ADM → INT 32h

Or any layout you prefer.

################################
# Using INT to emulare in a DOS like box, the major CAST and POLL commands
################################
; -------------------------------
; INT 30h — CAST Dispatcher
; -------------------------------
CAST_DISPATCH:
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX
    PUSH SI
    PUSH DI

    MOV AL, [SP+14]     ; function ID pushed by EX716 code
                        ; (you can choose any ABI you like)

    CMP AL, 0
    JE  CAST_PRINT_STR

    CMP AL, 1
    JE  CAST_PRINT_INT

    CMP AL, 2
    JE  CAST_END

    CMP AL, 3
    JE  CAST_STACK_DUMP

    JMP CAST_DONE


CAST_PRINT_STR:
    POP BX              ; EX716 address
    PUSH BX             ; restore stack after use

    MOV DX, BX          ; DS:DX = pointer to string
    MOV AH, 09h
    INT 21h
    JMP CAST_DONE


CAST_PRINT_INT:
    POP AX              ; value to print
    PUSH AX             ; restore stack after use

    CALL PRINT_INT16
    JMP CAST_DONE


PRINT_INT16:
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX

    ; AX = number
    ; Convert to decimal (unsigned for now)
    XOR CX, CX          ; digit count
    MOV BX, 10

PI_LOOP:
    XOR DX, DX
    DIV BX              ; AX = AX/10, DX = remainder
    PUSH DX             ; save digit
    INC CX
    CMP AX, 0
    JNE PI_LOOP

PI_OUT:
    POP DX
    ADD DL, '0'
    MOV AH, 02h
    INT 21h
    LOOP PI_OUT

    POP DX
    POP CX
    POP BX
    POP AX
    RET


CAST_END:
    MOV AH, 4Ch
    MOV AL, 00h
    INT 21h


CAST_STACK_DUMP:
    MOV SI, SP          ; start at current SP
    MOV DI, STACK_BASE  ; end of EX716 stack

CSD_LOOP:
    CMP SI, DI
    JAE CAST_DONE

    MOV AX, [SI]
    CALL PRINT_HEX16

    MOV DL, 0Dh
    MOV AH, 02h
    INT 21h
    MOV DL, 0Ah
    INT 21h

    ADD SI, 2
    JMP CSD_LOOP


PRINT_HEX16:
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX

    MOV CX, 4
PH_LOOP:
    ROL AX, 4
    MOV DL, AL
    AND DL, 0Fh
    CMP DL, 9
    JBE PH_DIGIT
    ADD DL, 7
PH_DIGIT:
    ADD DL, '0'
    MOV AH, 02h
    INT 21h
    LOOP PH_LOOP

    POP DX
    POP CX
    POP BX
    POP AX
    RET


POLL_DISPATCH:
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX
    PUSH SI
    PUSH DI

    MOV AL, [SP+14]     ; function ID

    CMP AL, 0
    JE  POLL_READ_INTI

    CMP AL, 1
    JE  POLL_READ_STRI

    JMP POLL_DONE


POLL_READ_INTI:
    POP BX              ; EX716 address
    PUSH BX

    CALL READ_INT16
    MOV [BX], AX
    JMP POLL_DONE


READ_INT16:
    PUSH BX
    PUSH CX
    PUSH DX

    MOV AH, 0Ah
    MOV DX, OFFSET INPUT_BUFFER
    INT 21h

    ; parse decimal
    MOV SI, OFFSET INPUT_BUFFER+2
    XOR AX, AX
RI_LOOP:
    MOV DL, [SI]
    CMP DL, 0Dh
    JE  RI_DONE
    SUB DL, '0'
    MOV CX, AX
    MOV AX, 10
    MUL CX
    ADD AX, DX
    INC SI
    JMP RI_LOOP

RI_DONE:
    POP DX
    POP CX
    POP BX
    RET


POLL_READ_STRI:
    POP BX              ; EX716 address
    POP CX              ; max length
    PUSH CX
    PUSH BX

    MOV AH, 0Ah
    MOV DX, OFFSET INPUT_BUFFER
    INT 21h

    ; copy into EX716 memory
    MOV SI, OFFSET INPUT_BUFFER+2
    MOV DI, BX
    MOV CL, [INPUT_BUFFER+1] ; actual length
    REP MOVSB

    MOV BYTE PTR [DI], 0     ; zero-terminate
    JMP POLL_DONE


CAST_DONE:
POLL_DONE:
    POP DI
    POP SI
    POP DX
    POP CX
    POP BX
    POP AX
    IRET
